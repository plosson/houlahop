Install by hand 
# macOS and Linux
curl -LsSf https://houlahop.com/agentio/install | sh

# Windows (PowerShell)
iwr -useb https://houlahop.com/agentio/install.ps1 | iex

# Homebrew
brew tap plosson/agentio && brew install agentio
